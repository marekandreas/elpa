function(_elpa_extract_marked_lines output)
  set(_elpa_markers "${ARGV1}")
  set(_elpa_sources "${ARGV2}")
  file(WRITE "${output}" "")

  foreach(_elpa_source IN LISTS _elpa_sources)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_elpa_source}")
    file(STRINGS "${_elpa_source}" _elpa_lines)
    foreach(_elpa_line IN LISTS _elpa_lines)
      foreach(_elpa_marker IN LISTS _elpa_markers)
        if(_elpa_line MATCHES "^[ \t]*${_elpa_marker}")
          string(REGEX REPLACE "^[ \t]*${_elpa_marker}" "" _elpa_interface "${_elpa_line}")
          file(APPEND "${output}" "${_elpa_interface}\n")
        endif()
      endforeach()
    endforeach()
  endforeach()
endfunction()

# ELPA uses preprocessor-expanded Fortran templates. Asking CMake to
# preprocess every Fortran source before it scans module dependencies lets its
# native scanner see the modules used inside those templates. This avoids
# hand-maintained object or .mod dependency edges that are fragile with both
# Ninja dyndep and parallel Make builds.
set_source_files_properties(
  ${_elpa_sources}
  PROPERTIES Fortran_PREPROCESS ON
)

    endforeach()
  endfunction()
